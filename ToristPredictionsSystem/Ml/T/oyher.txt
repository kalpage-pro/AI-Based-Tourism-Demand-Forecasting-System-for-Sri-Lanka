import json
import math
import joblib
import pandas as pd
import streamlit as st
from pathlib import Path

st.set_page_config(page_title="Tourism Prediction UI", layout="centered")
st.title("Tourism ML – Predict using Year + Month + Dollar Rate")

# ---------- Locate metadata.json ----------
APP_DIR = Path(__file__).parent if "__file__" in globals() else Path.cwd()
CANDIDATES = [APP_DIR / "metadata.json", APP_DIR / "models" / "metadata.json"]
META_PATH = next((p for p in CANDIDATES if p.exists()), None)

if META_PATH is None:
    st.error("metadata.json not found (expected beside app.py or inside models/).")
    st.stop()

BASE_DIR = META_PATH.parent  # where metadata lives
with open(META_PATH, "r", encoding="utf-8") as f:
    meta = json.load(f)

FEATURES_BASE = meta["features_base"]
FEATURES_REV = meta["features_revenue"]
MODEL_FILES = meta["model_files"]

# Optional: defaults (if your metadata contains it)
DEFAULTS = meta.get("defaults", {})  # dict like {"marketing_spend": 1200, ...}

TARGET_OPTIONS = {
    "Arrivals (total_arrivals)": ("arrivals", FEATURES_BASE, ["rf_arrivals", "xgb_arrivals"]),
    "Revenue (tourism_revenue)": ("revenue", FEATURES_REV, ["rf_revenue", "xgb_revenue"]),
    "Occupancy (hotel_occupancy_rate)": ("occupancy", FEATURES_BASE, ["rf_occupancy", "xgb_occupancy"]),
}

# ---------- Select target + model ----------
target_label = st.selectbox("What do you want to predict?", list(TARGET_OPTIONS.keys()))
_, required_features, allowed_models = TARGET_OPTIONS[target_label]
model_key = st.selectbox("Choose model", allowed_models)

raw_model_path = Path(MODEL_FILES[model_key])
model_path = raw_model_path if raw_model_path.is_absolute() else (BASE_DIR / raw_model_path)

# extra fallback: sometimes models are in same folder as app.py
if not model_path.exists():
    alt = APP_DIR / raw_model_path.name
    if alt.exists():
        model_path = alt

if not model_path.exists():
    st.error(f"Model file not found: {model_path}")
    st.stop()

model = joblib.load(model_path)
st.caption(f"Loaded: {model_key} from {model_path}")

# ---------- Inputs (ONLY 3) ----------
st.subheader("Inputs")
year = st.number_input("year", min_value=1900, max_value=2100, value=2025, step=1)
month_num = st.number_input("month_num (1-12)", min_value=1, max_value=12, value=1, step=1)

# IMPORTANT: adjust this feature name to the one in your dataset
# If your feature name is different (e.g., "usd_lkr_rate", "exchange_rate"), change below.
usd_feature_name = None
for candidate in ["usd_lkr", "dollar_rate", "exchange_rate", "usd_rate", "usd_to_lkr"]:
    if candidate in required_features:
        usd_feature_name = candidate
        break

usd_lkr = st.number_input("Dollar Rate (USD→LKR)", min_value=0.0, value=300.0, step=1.0)

# Derived time features
quarter = (int((month_num - 1) // 3) + 1)
month_sin = math.sin(2 * math.pi * month_num / 12)
month_cos = math.cos(2 * math.pi * month_num / 12)

# ---------- Build full feature row ----------
row = {}

for feat in required_features:
    if feat == "year":
        row[feat] = float(year)
    elif feat == "month_num":
        row[feat] = float(month_num)
    elif feat == "quarter":
        row[feat] = float(quarter)
    elif feat == "month_sin":
        row[feat] = float(month_sin)
    elif feat == "month_cos":
        row[feat] = float(month_cos)
    elif usd_feature_name and feat == usd_feature_name:
        row[feat] = float(usd_lkr)
    else:
        # Fill missing features using defaults from metadata, else 0
        row[feat] = float(DEFAULTS.get(feat, 0.0))

X = pd.DataFrame([row], columns=required_features)

st.subheader("Final model input (auto-filled)")
st.dataframe(X, use_container_width=True)

# ---------- Predict ----------
if st.button("Predict"):
    try:
        pred = model.predict(X)[0]
        st.success(f"✅ Prediction: {pred}")
    except Exception as e:
        st.error(f"Prediction failed: {e}")

st.caption("Note: Only Year/Month/Dollar Rate are user inputs. Other required features are auto-filled.")
